# 退出并删除旧环境
conda deactivate
conda remove -n paddlelabel --all -y

# 创建新环境
conda create -n paddlelabel python=3.11 -y
conda activate paddlelabel

# 先安装关键依赖
pip install marshmallow==3.19.0
pip install connexion==2.14.1
pip install Flask==2.2.5
pip install Werkzeug==2.2.2
pip install a2wsgi==1.8.0
pip install uvicorn==0.18.1

# 最后安装 PaddleLabel
pip install paddlelabel

# 启动
paddlelabel
